```python
import json
import numpy as np
import pandas as pd

NY = "America/New_York"
REQUIRED = ["record_id", "tpep_pickup_datetime", "trip_distance", "payment_type", "fare_amount"]
CONTRACT_COLS = [
    "record_id", "VendorID", "tpep_pickup_datetime", "tpep_dropoff_datetime",
    "PULocationID", "DOLocationID", "trip_distance", "payment_type",
    "fare_amount", "tip_amount", "total_amount",
]
NUMERIC_OPTIONAL = ["VendorID", "PULocationID", "DOLocationID", "tip_amount", "total_amount"]


class Abstain(Exception):
    pass


def _abstain(reason):
    return {"status": "abstain", "metrics": None, "reason": reason}


def _to_ny_naive(s):
    """Convert a series/index-like to naive local New York datetimes."""
    if isinstance(s, pd.Series) and pd.api.types.is_datetime64_any_dtype(s.dtype):
        res = s
    else:
        res = None
        if s.dtype == object or pd.api.types.is_string_dtype(s.dtype):
            try:
                res = pd.to_datetime(s, format="ISO8601")
                if res.dtype == object:
                    res = None
            except Exception:
                res = None
            if res is None:
                res = pd.to_datetime(s, utc=True, format="ISO8601")
        else:
            res = pd.to_datetime(s)
    if getattr(res.dtype, "tz", None) is not None:
        res = res.dt.tz_convert(NY).dt.tz_localize(None)
    return res.astype("datetime64[ns]")


def _to_numeric(s, name):
    try:
        out = pd.to_numeric(s, errors="raise")
    except Exception:
        raise Abstain("column %s has non-numeric values" % name)
    if out.dtype == bool:
        raise Abstain("column %s is boolean" % name)
    return out.astype("float64") if not pd.api.types.is_integer_dtype(out.dtype) else out


def _parse_period(manifest):
    rp = manifest.get("reporting_period")
    if rp is None:
        raise Abstain("manifest lacks reporting_period")
    start = end = None
    if isinstance(rp, dict):
        for k in ("start", "from", "begin", "start_date"):
            if k in rp:
                start = rp[k]
                break
        for k in ("end", "to", "stop", "end_date"):
            if k in rp:
                end = rp[k]
                break
    elif isinstance(rp, (list, tuple)) and len(rp) == 2:
        start, end = rp
    if start is None or end is None:
        raise Abstain("cannot parse reporting_period")
    try:
        st = pd.Timestamp(start)
        en = pd.Timestamp(end)
        if st.tzinfo is not None:
            st = st.tz_convert(NY).tz_localize(None)
        if en.tzinfo is not None:
            en = en.tz_convert(NY).tz_localize(None)
    except Exception:
        raise Abstain("unparseable reporting_period")
    if not st < en:
        raise Abstain("invalid reporting_period")
    return st, en


def _analyze(data_path, manifest_path):
    try:
        with open(manifest_path, "r", encoding="utf-8") as f:
            manifest = json.load(f)
    except Exception:
        raise Abstain("cannot read manifest")
    if not isinstance(manifest, dict):
        raise Abstain("manifest malformed")
    start, end = _parse_period(manifest)

    try:
        df = pd.read_parquet(data_path)
    except Exception:
        raise Abstain("cannot read data file")

    # renames
    changes = manifest.get("schema_changes") or []
    if isinstance(changes, list):
        for ch in changes:
            if not isinstance(ch, dict) or ch.get("type") != "rename":
                continue
            src, dst = ch.get("from"), ch.get("to")
            if src is None or dst is None:
                continue
            if src in df.columns and dst in df.columns and src != dst:
                raise Abstain("both original and renamed column present: %s/%s" % (src, dst))
            if src in df.columns and src != dst:
                df = df.rename(columns={src: dst})

    if df.columns.duplicated().any():
        raise Abstain("duplicate column names")
    missing = [c for c in REQUIRED if c not in df.columns]
    if missing:
        raise Abstain("missing required columns: %s" % ",".join(missing))

    d = pd.DataFrame(index=df.index)
    for c in REQUIRED:
        if df[c].isna().any():
            raise Abstain("nulls in required column %s" % c)

    try:
        d["tpep_pickup_datetime"] = _to_ny_naive(df["tpep_pickup_datetime"])
    except Exception:
        raise Abstain("unparseable pickup datetime")
    if "tpep_dropoff_datetime" in df.columns:
        try:
            d["tpep_dropoff_datetime"] = _to_ny_naive(df["tpep_dropoff_datetime"])
        except Exception:
            raise Abstain("unparseable dropoff datetime")
    if d["tpep_pickup_datetime"].isna().any():
        raise Abstain("null pickup datetime")

    for c in ["record_id", "trip_distance", "payment_type", "fare_amount"]:
        d[c] = _to_numeric(df[c], c)
    for c in NUMERIC_OPTIONAL:
        if c in df.columns:
            d[c] = _to_numeric(df[c], c)

    for c in ["record_id", "payment_type"]:
        v = d[c].to_numpy(dtype="float64")
        if not np.isfinite(v).all() or (v != np.floor(v)).any():
            raise Abstain("non-integer values in %s" % c)
    for c in ["trip_distance", "fare_amount"]:
        if not np.isfinite(d[c].to_numpy(dtype="float64")).all():
            raise Abstain("non-finite values in %s" % c)
    d["record_id"] = d["record_id"].astype("int64")
    d["payment_type"] = d["payment_type"].astype("int64")

    # dedup / conflicts
    cols = [c for c in CONTRACT_COLS if c in d.columns]
    d = d[cols].drop_duplicates()
    if d["record_id"].duplicated().any():
        raise Abstain("conflicting duplicate record_id values")

    # validity
    if not ((d["trip_distance"] > 0) & (d["trip_distance"] <= 100)).all():
        raise Abstain("trip_distance outside valid range")
    if not ((d["fare_amount"] >= 0) & (d["fare_amount"] <= 1000)).all():
        raise Abstain("fare_amount outside valid range")
    if not d["payment_type"].isin([0, 1, 2, 3, 4, 5, 6]).all():
        raise Abstain("invalid payment_type code")
    pu = d["tpep_pickup_datetime"]
    if not ((pu >= start) & (pu < end)).all():
        raise Abstain("trips outside reporting period")

    n = len(d)
    if n == 0:
        raise Abstain("no trips")
    fares = d["fare_amount"].to_numpy(dtype="float64")
    total = float(np.sum(fares))
    return {
        "status": "ok",
        "metrics": {
            "trip_count": int(n),
            "total_fare": total,
            "mean_fare": total / n,
            "mean_trip_distance": float(np.mean(d["trip_distance"].to_numpy(dtype="float64"))),
            "card_share": float((d["payment_type"] == 1).sum()) / n,
        },
        "reason": "",
    }


def analyze(data_path: str, manifest_path: str) -> dict:
    try:
        return _analyze(data_path, manifest_path)
    except Abstain as e:
        return _abstain(str(e))
    except Exception as e:
        return _abstain("unexpected error: %s" % type(e).__name__)
```
