"""NYC TLC yellow taxi monthly summary with data-contract verification.

analyze(data_path, manifest_path) runs the full verification protocol before
computing metrics. Any failure returns an abstain result with a reason.
Nothing is printed or written.
"""

import hashlib
import json
import math
import os
import re
import warnings

import numpy as np
import pandas as pd
import pyarrow as pa
import pyarrow.parquet as pq

_TZ = "America/New_York"

_CONTRACT = {
    "record_id": "int",
    "VendorID": "int",
    "tpep_pickup_datetime": "ts",
    "tpep_dropoff_datetime": "ts",
    "PULocationID": "int",
    "DOLocationID": "int",
    "trip_distance": "float",
    "payment_type": "int",
    "fare_amount": "float",
    "tip_amount": "float",
    "total_amount": "float",
}
_REQUIRED = ("record_id", "tpep_pickup_datetime", "trip_distance", "payment_type", "fare_amount")
_VALID_PAYMENT = (0, 1, 2, 3, 4, 5, 6)
_MANIFEST_KEYS = (
    "dataset",
    "snapshot_id",
    "snapshot_version",
    "reporting_period",
    "file",
    "file_sha256",
    "row_count",
    "columns",
    "units",
    "schema_changes",
    "description",
)
_SHA_RE = re.compile(r"[0-9a-fA-F]{64}\Z")


class _Abstain(Exception):
    """Internal signal that verification failed; carries the reason."""


def _fail(reason):
    raise _Abstain(reason)


def _local_naive(ts):
    if ts.tzinfo is not None:
        ts = ts.tz_convert(_TZ).tz_localize(None)
    return ts


def _parse_bound(value):
    if not isinstance(value, str):
        _fail("reporting_period bound is not a string")
    try:
        ts = pd.Timestamp(value)
    except (ValueError, TypeError):
        _fail("reporting_period bound is not a valid timestamp")
    if pd.isna(ts):
        _fail("reporting_period bound is missing")
    return _local_naive(ts)


def _read_manifest(manifest_path):
    if not isinstance(manifest_path, str) or not os.path.isfile(manifest_path):
        _fail("manifest not found")
    try:
        with open(manifest_path, "r", encoding="utf-8") as fh:
            manifest = json.load(fh)
    except (OSError, ValueError):
        _fail("manifest is not valid JSON")
    if not isinstance(manifest, dict):
        _fail("manifest is not a JSON object")
    missing = [k for k in _MANIFEST_KEYS if k not in manifest]
    if missing:
        _fail("manifest missing fields: " + ", ".join(missing))
    return manifest


def _check_manifest(m):
    for key in ("dataset", "snapshot_id", "file"):
        if not isinstance(m[key], str) or not m[key].strip():
            _fail("manifest field %s is malformed" % key)
    if m["snapshot_version"] is None:
        _fail("manifest snapshot_version is missing")

    rp = m["reporting_period"]
    if not isinstance(rp, dict) or any(k not in rp for k in ("start", "end", "timezone")):
        _fail("manifest reporting_period is malformed")
    if rp["timezone"] != _TZ:
        _fail("reporting_period timezone is not America/New_York")
    start = _parse_bound(rp["start"])
    end = _parse_bound(rp["end"])
    if start >= end:
        _fail("reporting_period start is not before end")

    sha = m["file_sha256"]
    if not isinstance(sha, str) or not _SHA_RE.match(sha):
        _fail("file_sha256 is not a SHA-256 hex digest")

    rc = m["row_count"]
    if isinstance(rc, bool) or not isinstance(rc, int) or rc < 0:
        _fail("row_count is malformed")

    cols = m["columns"]
    if not isinstance(cols, list) or not cols or not all(
        isinstance(c, dict) and isinstance(c.get("name"), str) and "type" in c for c in cols
    ):
        _fail("manifest columns are malformed")
    names = [c["name"] for c in cols]
    if len(set(names)) != len(names):
        _fail("manifest columns contain duplicate names")

    changes = m["schema_changes"]
    if not isinstance(changes, list):
        _fail("schema_changes is malformed")

    units = m["units"]
    if (
        not isinstance(units, dict)
        or str(units.get("trip_distance", "")).strip().lower() != "miles"
        or str(units.get("fare_amount", "")).strip().lower() != "usd"
    ):
        _fail("units do not match contract (trip_distance in miles, fare_amount in USD)")

    return {
        "start": start,
        "end": end,
        "sha": sha.lower(),
        "row_count": rc,
        "names": set(names),
        "changes": changes,
    }


def _map_columns(schema, changes, manifest_names):
    rename = {}
    for ch in changes:
        if not (
            isinstance(ch, dict)
            and ch.get("type") == "rename"
            and isinstance(ch.get("from"), str)
            and isinstance(ch.get("to"), str)
        ):
            _fail("unsupported schema change in manifest")
        if ch["from"] in rename:
            _fail("duplicate rename in manifest")
        rename[ch["from"]] = ch["to"]

    delivered = [f.name for f in schema]
    if len(set(delivered)) != len(delivered):
        _fail("duplicate column names in data file")
    for src in rename:
        if src not in delivered:
            _fail("renamed column %s not present in data file" % src)

    normalized = [rename.get(n, n) for n in delivered]
    if len(set(normalized)) != len(normalized):
        _fail("column name collision after applying schema_changes")
    if set(normalized) != manifest_names and set(delivered) != manifest_names:
        _fail("delivered columns disagree with manifest")

    index = {}
    for i, name in enumerate(normalized):
        if name in _CONTRACT:
            index[name] = i
    missing = [c for c in _CONTRACT if c not in index]
    if missing:
        _fail("data file missing contract columns: " + ", ".join(missing))
    return index


def _type_ok(t, kind):
    is_str = pa.types.is_string(t) or pa.types.is_large_string(t)
    if kind == "int":
        return pa.types.is_integer(t) or pa.types.is_floating(t) or is_str
    if kind == "float":
        return pa.types.is_integer(t) or pa.types.is_floating(t) or is_str
    return pa.types.is_timestamp(t) or is_str


def _strip_strings(s):
    if s.dtype == object:
        return s.map(lambda v: v.strip() if isinstance(v, str) else v)
    return s


def _to_int(s):
    num = pd.to_numeric(_strip_strings(s), errors="raise")
    if not pd.api.types.is_integer_dtype(num):
        vals = num.to_numpy(dtype="float64")
        if not (np.isfinite(vals) | np.isnan(vals)).all():
            _fail("non-finite value in integer column")
        present = vals[~np.isnan(vals)]
        if not np.equal(np.floor(present), present).all():
            _fail("non-integer value in integer column")
    return num.astype("Int64")


def _to_float(s):
    num = pd.to_numeric(_strip_strings(s), errors="raise").astype("float64")
    if np.isinf(num.to_numpy()).any():
        _fail("non-finite value in numeric column")
    return num


def _to_ts(s):
    s = _strip_strings(s)
    try:
        with warnings.catch_warnings():
            warnings.simplefilter("ignore")
            out = pd.to_datetime(s, errors="raise")
    except (ValueError, TypeError, OverflowError):
        _fail("timestamp column is not parseable")
    if not pd.api.types.is_datetime64_any_dtype(out):
        _fail("timestamp column has mixed or unparseable zones")
    if getattr(out.dt, "tz", None) is not None:
        out = out.dt.tz_convert(_TZ).dt.tz_localize(None)
    return out


def _analyze(data_path, manifest_path):
    # Step 1: manifest.
    manifest = _read_manifest(manifest_path)
    info = _check_manifest(manifest)

    if not isinstance(data_path, str) or not os.path.isfile(data_path):
        _fail("data file not found")

    # Step 2: SHA-256 of the delivered bytes.
    digest = hashlib.sha256()
    with open(data_path, "rb") as fh:
        for chunk in iter(lambda: fh.read(1 << 20), b""):
            digest.update(chunk)
    if digest.hexdigest() != info["sha"]:
        _fail("file SHA-256 does not match manifest")

    # Step 3: row count.
    try:
        md = pq.read_metadata(data_path)
        schema = pq.read_schema(data_path)
    except Exception:
        _fail("data file is not readable Parquet")
    if md.num_rows != info["row_count"]:
        _fail("row count does not match manifest")

    # Step 4: schema agreement after schema_changes.
    index = _map_columns(schema, info["changes"], info["names"])
    for name, idx in index.items():
        if not _type_ok(schema.field(idx).type, _CONTRACT[name]):
            _fail("column %s has an incompatible type" % name)

    try:
        table = pq.read_table(data_path)
    except Exception:
        _fail("data file could not be read")

    frame = {}
    for name, idx in index.items():
        raw = pd.Series(table.column(idx).to_pandas())
        kind = _CONTRACT[name]
        if kind == "int":
            frame[name] = _to_int(raw)
        elif kind == "float":
            frame[name] = _to_float(raw)
        else:
            frame[name] = _to_ts(raw)
    df = pd.DataFrame(frame)

    # Step 6 (required fields and ranges), evaluated before period filtering.
    for name in _REQUIRED:
        if df[name].isna().any():
            _fail("null in required column %s" % name)

    dist = df["trip_distance"]
    if not ((dist > 0) & (dist <= 100)).all():
        _fail("trip_distance outside valid range (0, 100]")

    fare = df["fare_amount"]
    if not ((fare >= 0) & (fare <= 1000)).all():
        _fail("fare_amount outside valid range [0, 1000]")

    if not df["payment_type"].isin(list(_VALID_PAYMENT)).all():
        _fail("undocumented payment_type code")

    # Step 5: pickups inside the reporting period (start inclusive, end exclusive).
    pickup = df["tpep_pickup_datetime"]
    if ((pickup < info["start"]) | (pickup >= info["end"])).any():
        _fail("pickup timestamp outside reporting_period")

    # Step 6 (replays and conflicts).
    dedup = df.drop_duplicates()
    if dedup["record_id"].duplicated().any():
        _fail("conflicting records share a record_id")

    n = len(dedup)
    if n == 0:
        _fail("no trips to summarize")

    fares = dedup["fare_amount"].tolist()
    dists = dedup["trip_distance"].tolist()
    total_fare = math.fsum(fares)
    mean_fare = total_fare / n
    mean_distance = math.fsum(dists) / n
    card_trips = int((dedup["payment_type"] == 1).sum())
    card_share = card_trips / n

    return {
        "status": "ok",
        "metrics": {
            "trip_count": n,
            "total_fare": total_fare,
            "mean_fare": mean_fare,
            "mean_trip_distance": mean_distance,
            "card_share": card_share,
        },
        "reason": "",
    }


def analyze(data_path: str, manifest_path: str) -> dict:
    try:
        return _analyze(data_path, manifest_path)
    except _Abstain as exc:
        return {"status": "abstain", "metrics": None, "reason": str(exc)}
    except Exception as exc:
        return {
            "status": "abstain",
            "metrics": None,
            "reason": "unable to verify data: %s" % type(exc).__name__,
        }
