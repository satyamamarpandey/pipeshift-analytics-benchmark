```python
import hashlib
import json
import math
import os
import re
from decimal import Decimal

import numpy as np
import pandas as pd
import pyarrow.parquet as pq

NY_TZ = "America/New_York"

CONTRACT_COLUMNS = [
    "record_id", "VendorID", "tpep_pickup_datetime", "tpep_dropoff_datetime",
    "PULocationID", "DOLocationID", "trip_distance", "payment_type",
    "fare_amount", "tip_amount", "total_amount",
]
REQUIRED = ["record_id", "tpep_pickup_datetime", "trip_distance", "payment_type", "fare_amount"]
NUMERIC_COLS = ["record_id", "VendorID", "PULocationID", "DOLocationID", "trip_distance",
                "payment_type", "fare_amount", "tip_amount", "total_amount"]
INTEGER_COLS = ["record_id", "payment_type"]
DATETIME_COLS = ["tpep_pickup_datetime", "tpep_dropoff_datetime"]
VALID_PAYMENT = {0, 1, 2, 3, 4, 5, 6}
MANIFEST_FIELDS = ["dataset", "snapshot_id", "snapshot_version", "reporting_period", "file",
                   "file_sha256", "row_count", "columns", "units", "schema_changes",
                   "description"]

_AWARE_RE = re.compile(r"(Z|[+-]\d{2}:?\d{2})$", re.IGNORECASE)

_DIST_FACTORS = {
    "mi": 1.0, "mile": 1.0, "miles": 1.0,
    "km": 1 / 1.609344, "kilometer": 1 / 1.609344, "kilometers": 1 / 1.609344,
    "kilometre": 1 / 1.609344, "kilometres": 1 / 1.609344,
    "m": 1 / 1609.344, "meter": 1 / 1609.344, "meters": 1 / 1609.344,
    "metre": 1 / 1609.344, "metres": 1 / 1609.344,
}
_MONEY_FACTORS = {
    "usd": 1.0, "$": 1.0, "dollar": 1.0, "dollars": 1.0, "us dollars": 1.0, "usd dollars": 1.0,
    "cents": 0.01, "cent": 0.01, "usd cents": 0.01, "us cents": 0.01,
}


class _Abstain(Exception):
    pass


def _abstain(reason):
    return {"status": "abstain", "metrics": None, "reason": reason}


def _sha256(path):
    h = hashlib.sha256()
    with open(path, "rb") as f:
        for chunk in iter(lambda: f.read(1 << 20), b""):
            h.update(chunk)
    return h.hexdigest()


def _blank_to_none(v):
    if isinstance(v, str):
        v = v.strip()
        if v == "" or v.lower() in ("nan", "null", "none", "nat"):
            return None
    return v


def _to_numeric(s, name):
    if pd.api.types.is_bool_dtype(s):
        raise _Abstain(f"column {name} has boolean values")
    if pd.api.types.is_numeric_dtype(s):
        out = s.astype("float64")
    else:
        obj = s.astype(object).map(_blank_to_none)
        obj = obj.map(lambda v: float(v) if isinstance(v, Decimal) else v)
        if obj.map(lambda v: isinstance(v, bool)).any():
            raise _Abstain(f"column {name} has boolean values")
        out = pd.to_numeric(obj, errors="coerce").astype("float64")
        bad = obj.notna() & out.isna()
        if bad.any():
            raise _Abstain(f"column {name} has unparseable numeric values")
    return out


def _parse_one(v):
    v = _blank_to_none(v)
    if v is None or (isinstance(v, float) and math.isnan(v)):
        return pd.NaT
    ts = pd.Timestamp(v)
    if ts is pd.NaT:
        return pd.NaT
    if ts.tzinfo is not None:
        ts = ts.tz_convert(NY_TZ).tz_localize(None)
    return ts


def _to_local_datetime(s, name):
    if isinstance(s.dtype, pd.DatetimeTZDtype):
        return s.dt.tz_convert(NY_TZ).dt.tz_localize(None).astype("datetime64[ns]")
    if pd.api.types.is_datetime64_dtype(s):
        return s.astype("datetime64[ns]")
    if pd.api.types.is_bool_dtype(s):
        raise _Abstain(f"column {name} has boolean values")
    if pd.api.types.is_numeric_dtype(s):
        vals = s.astype("float64")
        nonnull = vals.dropna()
        if len(nonnull) == 0:
            return pd.Series(pd.NaT, index=s.index, dtype="datetime64[ns]")
        mx = nonnull.abs().max()
        unit = "ns" if mx > 1e17 else "us" if mx > 1e14 else "ms" if mx > 1e11 else "s"
        out = pd.to_datetime(vals, unit=unit, utc=True, errors="coerce")
        if (vals.notna() & out.isna()).any():
            raise _Abstain(f"column {name} has unparseable timestamps")
        return out.dt.tz_convert(NY_TZ).dt.tz_localize(None).astype("datetime64[ns]")

    obj = s.astype(object).map(_blank_to_none)
    result = pd.Series(pd.NaT, index=s.index, dtype="datetime64[ns]")
    is_str = obj.map(lambda v: isinstance(v, str))
    strs = obj[is_str]
    if len(strs):
        aware = strs.map(lambda v: bool(_AWARE_RE.search(v)))
        a = strs[aware]
        n = strs[~aware]
        if len(a):
            pa_ = pd.to_datetime(a, utc=True, errors="coerce")
            pa_ = pa_.dt.tz_convert(NY_TZ).dt.tz_localize(None)
            result.loc[a.index] = pa_.astype("datetime64[ns]")
        if len(n):
            pn = pd.to_datetime(n, errors="coerce")
            if isinstance(pn.dtype, pd.DatetimeTZDtype):
                pn = pn.dt.tz_convert(NY_TZ).dt.tz_localize(None)
            result.loc[n.index] = pn.astype("datetime64[ns]")
    # elementwise fallback for anything not yet parsed
    pending = obj.notna() & result.isna()
    for idx in obj.index[pending]:
        try:
            result.loc[idx] = _parse_one(obj.loc[idx])
        except Exception:
            raise _Abstain(f"column {name} has unparseable timestamps")
    if (obj.notna() & result.isna()).any():
        raise _Abstain(f"column {name} has unparseable timestamps")
    return result


def _period_bound(value, tz):
    ts = pd.Timestamp(value)
    if ts is pd.NaT:
        raise _Abstain("invalid reporting_period bound")
    if ts.tzinfo is None:
        ts = ts.tz_localize(tz)
    return ts.tz_convert(NY_TZ).tz_localize(None)


def _unit_factor(units, col, table, default_key):
    if not isinstance(units, dict):
        return 1.0
    u = units.get(col)
    if u is None:
        return 1.0
    if isinstance(u, dict):
        u = u.get("unit", u.get("units", default_key))
    if not isinstance(u, str):
        raise _Abstain(f"unrecognized unit for {col}")
    key = u.strip().lower()
    if key not in table:
        raise _Abstain(f"unsupported unit for {col}: {u}")
    return table[key]


def _run(data_path, manifest_path):
    # 1. manifest
    if not os.path.isfile(manifest_path):
        raise _Abstain("manifest missing")
    try:
        with open(manifest_path, "r", encoding="utf-8") as f:
            manifest = json.load(f)
    except Exception:
        raise _Abstain("manifest is not valid JSON")
    if not isinstance(manifest, dict):
        raise _Abstain("manifest is not a JSON object")
    missing = [k for k in MANIFEST_FIELDS if k not in manifest]
    if missing:
        raise _Abstain(f"manifest missing fields: {missing}")
    rp = manifest["reporting_period"]
    if not isinstance(rp, dict) or not all(k in rp for k in ("start", "end", "timezone")):
        raise _Abstain("manifest reporting_period malformed")
    sha = manifest["file_sha256"]
    if not isinstance(sha, str) or not re.fullmatch(r"[0-9a-fA-F]{64}", sha.strip()):
        raise _Abstain("manifest file_sha256 malformed")
    rc = manifest["row_count"]
    if isinstance(rc, bool) or not isinstance(rc, int) or rc < 0:
        raise _Abstain("manifest row_count malformed")
    cols = manifest["columns"]
    if not isinstance(cols, list) or not cols or not all(
            isinstance(c, dict) and isinstance(c.get("name"), str) for c in cols):
        raise _Abstain("manifest columns malformed")
    changes = manifest["schema_changes"]
    if changes is None:
        changes = []
    if not isinstance(changes, list):
        raise _Abstain("manifest schema_changes malformed")
    renames = {}
    for ch in changes:
        if not isinstance(ch, dict):
            raise _Abstain("manifest schema_changes malformed")
        if ch.get("type") == "rename":
            fr, to = ch.get("from"), ch.get("to")
            if not isinstance(fr, str) or not isinstance(to, str):
                raise _Abstain("malformed rename in schema_changes")
            renames[fr] = to

    # 2. checksum
    if not os.path.isfile(data_path):
        raise _Abstain("data file missing")
    if _sha256(data_path).lower() != sha.strip().lower():
        raise _Abstain("data file SHA-256 does not match manifest")

    # 3. row count
    try:
        table = pq.read_table(data_path)
    except Exception:
        raise _Abstain("data file is not readable Parquet")
    if table.num_rows != rc:
        raise _Abstain(f"row count {table.num_rows} != manifest row_count {rc}")
    try:
        df = table.to_pandas()
    except Exception:
        raise _Abstain("could not convert data to pandas")
    if len(df) != rc:
        raise _Abstain("row count mismatch")

    # 4. columns
    delivered = [str(c) for c in df.columns]
    if len(set(delivered)) != len(delivered):
        raise _Abstain("duplicate column names in data")
    mnames = [c["name"] for c in cols]
    mapped = [renames.get(n, n) for n in mnames]
    if set(delivered) != set(mapped) and set(delivered) != set(mnames):
        raise _Abstain("delivered columns do not match manifest after schema_changes")
    df.columns = delivered
    inverse = {}
    for fr, to in renames.items():
        if to in df.columns and fr not in df.columns:
            inverse[to] = fr
    # resolve chained renames
    df = df.rename(columns=inverse)
    changed = True
    while changed:
        changed = False
        for fr, to in renames.items():
            if to in df.columns and fr not in df.columns:
                df = df.rename(columns={to: fr})
                changed = True
    miss = [c for c in CONTRACT_COLUMNS if c not in df.columns]
    if miss:
        raise _Abstain(f"contract columns missing: {miss}")
    if df.columns.duplicated().any():
        raise _Abstain("duplicate columns after normalization")

    # normalize types
    for c in NUMERIC_COLS:
        df[c] = _to_numeric(df[c], c)
    for c in DATETIME_COLS:
        df[c] = _to_local_datetime(df[c], c)

    # units
    units = manifest["units"]
    dfac = _unit_factor(units, "trip_distance", _DIST_FACTORS, "miles")
    if dfac != 1.0:
        df["trip_distance"] = df["trip_distance"] * dfac
    for c in ("fare_amount", "tip_amount", "total_amount"):
        mfac = _unit_factor(units, c, _MONEY_FACTORS, "usd")
        if mfac != 1.0:
            df[c] = df[c] * mfac

    # required non-null
    for c in REQUIRED:
        if df[c].isna().any():
            raise _Abstain(f"required column {c} has nulls")

    # 5. reporting period
    try:
        tz = rp["timezone"] or NY_TZ
        start = _period_bound(rp["start"], tz)
        end = _period_bound(rp["end"], tz)
    except _Abstain:
        raise
    except Exception:
        raise _Abstain("invalid reporting_period")
    if not start < end:
        raise _Abstain("reporting_period start not before end")
    pu = df["tpep_pickup_datetime"]
    if ((pu < start) | (pu >= end)).any():
        raise _Abstain("pickup timestamps outside reporting_period")

    # 6. contract rules
    for c in INTEGER_COLS:
        v = df[c]
        if (~np.isfinite(v)).any() or (v != np.round(v)).any():
            raise _Abstain(f"column {c} has non-integer values")
    if not df["payment_type"].isin(list(VALID_PAYMENT)).all():
        raise _Abstain("undocumented payment_type codes")
    td = df["trip_distance"]
    if not ((td > 0) & (td <= 100)).all():
        raise _Abstain("trip_distance out of valid range")
    fa = df["fare_amount"]
    if not ((fa >= 0) & (fa <= 1000)).all():
        raise _Abstain("fare_amount out of valid range")

    # replays and conflicts
    try:
        dedup = df.drop_duplicates(keep="first")
    except TypeError:
        dedup = df.loc[~df.astype(str).duplicated(keep="first")]
    if dedup["record_id"].duplicated().any():
        raise _Abstain("conflicting records share record_id")
    if len(dedup) == 0:
        raise _Abstain("no trips in data")

    n = int(len(dedup))
    fare = dedup["fare_amount"].astype("float64")
    total_fare = float(fare.sum())
    return {
        "trip_count": n,
        "total_fare": total_fare,
        "mean_fare": float(fare.mean()),
        "mean_trip_distance": float(dedup["trip_distance"].astype("float64").mean()),
        "card_share": float((dedup["payment_type"] == 1).sum() / n),
    }


def analyze(data_path: str, manifest_path: str) -> dict:
    try:
        metrics = _run(data_path, manifest_path)
    except _Abstain as e:
        return _abstain(str(e))
    except Exception as e:
        return _abstain(f"unexpected error during validation: {type(e).__name__}")
    return {"status": "ok", "metrics": metrics, "reason": ""}
```
