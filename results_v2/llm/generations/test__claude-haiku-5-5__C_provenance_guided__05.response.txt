```python
import hashlib
import json
import math
import os
import re
import warnings

import numpy as np
import pandas as pd
import pyarrow as pa
import pyarrow.parquet as pq

_TZ = "America/New_York"
_MANIFEST_KEYS = {
    "dataset", "snapshot_id", "snapshot_version", "reporting_period", "file",
    "file_sha256", "row_count", "columns", "units", "schema_changes", "description",
}
# Canonical data contract: column name -> kind
_CONTRACT = {
    "record_id": "int",
    "VendorID": "int",
    "tpep_pickup_datetime": "datetime",
    "tpep_dropoff_datetime": "datetime",
    "PULocationID": "int",
    "DOLocationID": "int",
    "trip_distance": "float",
    "payment_type": "int",
    "fare_amount": "float",
    "tip_amount": "float",
    "total_amount": "float",
}
_REQUIRED = {"record_id", "tpep_pickup_datetime", "trip_distance", "payment_type", "fare_amount"}
_PAYMENT_CODES = [0, 1, 2, 3, 4, 5, 6]
_SHA_RE = re.compile(r"^[0-9a-fA-F]{64}$")


class _Abstain(Exception):
    pass


def _abstain(reason):
    return {"status": "abstain", "metrics": None, "reason": reason}


def _load_manifest(path):
    if not isinstance(path, str) or not os.path.isfile(path):
        raise _Abstain("manifest file does not exist")
    try:
        with open(path, "r", encoding="utf-8") as fh:
            m = json.load(fh)
    except (OSError, UnicodeDecodeError, ValueError):
        raise _Abstain("manifest is not valid JSON")
    if not isinstance(m, dict):
        raise _Abstain("manifest root is not a JSON object")
    missing = _MANIFEST_KEYS - set(m.keys())
    if missing:
        raise _Abstain(f"manifest missing fields: {sorted(missing)}")
    return m


def _parse_bound(text, label):
    if not isinstance(text, str):
        raise _Abstain(f"reporting_period.{label} must be a string")
    try:
        ts = pd.Timestamp(text)
    except (ValueError, TypeError):
        raise _Abstain(f"reporting_period.{label} is not a valid timestamp")
    if pd.isna(ts):
        raise _Abstain(f"reporting_period.{label} is not a valid timestamp")
    if ts.tzinfo is not None:
        ts = ts.tz_convert(_TZ).tz_localize(None)
    return ts


def _as_int(s, name, nullable):
    if pd.api.types.is_bool_dtype(s):
        raise _Abstain(f"{name}: boolean values are not allowed")
    try:
        n = pd.to_numeric(s, errors="raise")
    except (ValueError, TypeError):
        raise _Abstain(f"{name}: values are not numeric")
    if pd.api.types.is_bool_dtype(n) or not pd.api.types.is_numeric_dtype(n):
        raise _Abstain(f"{name}: values are not numeric")
    if pd.api.types.is_integer_dtype(n):
        out = n.astype("Int64")
    else:
        f = n.astype("float64")
        vals = f.dropna().to_numpy()
        if not np.all(np.isfinite(vals)) or not np.all(vals == np.floor(vals)):
            raise _Abstain(f"{name}: non-integral or non-finite values")
        out = f.astype("Int64")
    if not nullable and out.isna().any():
        raise _Abstain(f"{name}: required field contains nulls")
    return out


def _as_float(s, name, nullable):
    if pd.api.types.is_bool_dtype(s):
        raise _Abstain(f"{name}: boolean values are not allowed")
    try:
        n = pd.to_numeric(s, errors="raise")
    except (ValueError, TypeError):
        raise _Abstain(f"{name}: values are not numeric")
    if pd.api.types.is_bool_dtype(n) or not pd.api.types.is_numeric_dtype(n):
        raise _Abstain(f"{name}: values are not numeric")
    f = n.astype("float64")
    if np.isinf(f.to_numpy()).any():
        raise _Abstain(f"{name}: infinite values")
    if not nullable and f.isna().any():
        raise _Abstain(f"{name}: required field contains nulls")
    return f


def _as_ts(s, name, nullable):
    if pd.api.types.is_datetime64_any_dtype(s):
        out = s
    else:
        try:
            out = pd.to_datetime(s, format="ISO8601", errors="raise")
        except (ValueError, TypeError):
            raise _Abstain(f"{name}: values are not valid timestamps")
        if not pd.api.types.is_datetime64_any_dtype(out):
            raise _Abstain(f"{name}: mixed timezone offsets are ambiguous")
    if getattr(out.dt, "tz", None) is not None:
        out = out.dt.tz_convert(_TZ).dt.tz_localize(None)
    if not nullable and out.isna().any():
        raise _Abstain(f"{name}: required field contains nulls")
    return out


def _analyze(data_path, manifest_path):
    # Step 1: manifest exists, is valid JSON, and has the required fields.
    m = _load_manifest(manifest_path)

    rp = m["reporting_period"]
    if not isinstance(rp, dict) or not {"start", "end", "timezone"} <= set(rp.keys()):
        raise _Abstain("reporting_period must contain start, end and timezone")
    if rp["timezone"] != _TZ:
        raise _Abstain("reporting_period timezone is not America/New_York")
    start = _parse_bound(rp["start"], "start")
    end = _parse_bound(rp["end"], "end")
    if not start < end:
        raise _Abstain("reporting_period start must precede end")

    if not isinstance(m["file_sha256"], str) or not _SHA_RE.match(m["file_sha256"]):
        raise _Abstain("file_sha256 is malformed")
    row_count = m["row_count"]
    if isinstance(row_count, bool) or not isinstance(row_count, int) or row_count < 0:
        raise _Abstain("row_count must be a non-negative integer")

    cols = m["columns"]
    if not isinstance(cols, list) or not all(
        isinstance(c, dict) and isinstance(c.get("name"), str) for c in cols
    ):
        raise _Abstain("manifest columns are malformed")
    mnames = [c["name"] for c in cols]
    if len(set(mnames)) != len(mnames):
        raise _Abstain("manifest columns contain duplicate names")
    if set(mnames) != set(_CONTRACT):
        raise _Abstain("manifest columns do not match the data contract")

    units = m["units"]
    if not isinstance(units, dict):
        raise _Abstain("units must be an object")
    if "mile" not in str(units.get("trip_distance", "")).lower():
        raise _Abstain("trip_distance units must be miles")
    fare_units = str(units.get("fare_amount", "")).lower()
    if "usd" not in fare_units and "dollar" not in fare_units:
        raise _Abstain("fare_amount units must be USD")

    changes = m["schema_changes"]
    if not isinstance(changes, list):
        raise _Abstain("schema_changes must be a list")
    rmap = {}
    for ch in changes:
        if not (
            isinstance(ch, dict)
            and ch.get("type") == "rename"
            and isinstance(ch.get("from"), str)
            and isinstance(ch.get("to"), str)
        ):
            raise _Abstain("unsupported schema change declaration")
        if ch["from"] not in mnames or ch["from"] in rmap:
            raise _Abstain("invalid rename declaration")
        rmap[ch["from"]] = ch["to"]

    delivered_expected = [rmap.get(n, n) for n in mnames]
    if len(set(delivered_expected)) != len(delivered_expected):
        raise _Abstain("rename collides with another column")
    canonical_of = {rmap.get(n, n): n for n in mnames}

    # Step 2: SHA-256 of the exact delivered bytes.
    if not isinstance(data_path, str) or not os.path.isfile(data_path):
        raise _Abstain("data file does not exist")
    with open(data_path, "rb") as fh:
        raw = fh.read()
    if hashlib.sha256(raw).hexdigest() != m["file_sha256"].strip().lower():
        raise _Abstain("SHA-256 of data file does not match manifest")

    try:
        table = pq.read_table(pa.BufferReader(raw))
    except Exception:
        raise _Abstain("data file is not readable Parquet")
    del raw

    # Step 3: row count.
    if table.num_rows != row_count:
        raise _Abstain("row count does not match manifest")

    # Step 4: columns agree with manifest after schema_changes.
    names = table.column_names
    if len(set(names)) != len(names) or set(names) != set(delivered_expected):
        raise _Abstain("delivered columns do not match manifest after schema_changes")

    df = table.to_pandas()
    df.columns = [canonical_of[c] for c in names]

    # Normalize types per contract.
    cols_out = {}
    for name, kind in _CONTRACT.items():
        s = df[name]
        nullable = name not in _REQUIRED
        if kind == "int":
            cols_out[name] = _as_int(s, name, nullable)
        elif kind == "float":
            cols_out[name] = _as_float(s, name, nullable)
        else:
            cols_out[name] = _as_ts(s, name, nullable)
    d = pd.DataFrame(cols_out)

    # Step 6 (domain rules), and step 5 (reporting period, checked on all delivered rows).
    dist = d["trip_distance"]
    if not ((dist > 0) & (dist <= 100)).all():
        raise _Abstain("trip_distance outside valid range (0, 100] miles")
    fare = d["fare_amount"]
    if not ((fare >= 0) & (fare <= 1000)).all():
        raise _Abstain("fare_amount outside valid range [0, 1000] USD")
    if not d["payment_type"].isin(_PAYMENT_CODES).all():
        raise _Abstain("payment_type contains undocumented codes")

    pick = d["tpep_pickup_datetime"]
    inside = (pick >= start) & (pick < end)
    if not inside.all():
        raise _Abstain("pickup timestamps outside reporting_period")

    # Replays (identical rows) count once; same record_id with different values is a conflict.
    d = d.drop_duplicates(keep="first").reset_index(drop=True)
    if d["record_id"].duplicated().any():
        raise _Abstain("conflicting duplicate record_id with differing values")

    n = len(d)
    if n == 0:
        raise _Abstain("no trips in reporting period")

    fares = d["fare_amount"].astype("float64").tolist()
    distances = d["trip_distance"].astype("float64").tolist()
    total_fare = math.fsum(fares)
    card_trips = int((d["payment_type"] == 1).sum())

    metrics = {
        "trip_count": int(n),
        "total_fare": float(total_fare),
        "mean_fare": float(total_fare / n),
        "mean_trip_distance": float(math.fsum(distances) / n),
        "card_share": float(card_trips / n),
    }
    return {"status": "ok", "metrics": metrics, "reason": ""}


def analyze(data_path: str, manifest_path: str) -> dict:
    try:
        with warnings.catch_warnings():
            warnings.simplefilter("ignore")
            return _analyze(data_path, manifest_path)
    except _Abstain as exc:
        return _abstain(str(exc))
    except Exception as exc:
        return _abstain(f"verification failed: {type(exc).__name__}: {exc}")
```
