import pandas as pd
import pyarrow.parquet as pq

def analyze(data_path: str, manifest_path: str) -> dict:
    try:
        # Load data
        data = pq.read_table(data_path).to_pandas()
        
        # Validate required columns
        required_columns = ['record_id', 'tpep_pickup_datetime', 'trip_distance', 'payment_type', 'fare_amount']
        if not all(column in data.columns for column in required_columns):
            return {"status": "abstain", "metrics": None, "reason": "Missing required columns"}
        
        # Remove null values for required columns
        data = data.dropna(subset=required_columns)
        
        # Filter data within reporting period
        manifest = pd.read_json(manifest_path)
        reporting_period = pd.Interval(pd.to_datetime(manifest['reporting_period']['start']), pd.to_datetime(manifest['reporting_period']['end']), closed='left')
        data = data[data['tpep_pickup_datetime'].between(reporting_period.left, reporting_period.right, inclusive='left')]
        
        # Remove replayed records
        data = data.drop_duplicates(subset=required_columns, keep='first')
        
        # Calculate metrics
        trip_count = len(data)
        total_fare = data['fare_amount'].sum()
        mean_fare = data['fare_amount'].mean()
        mean_trip_distance = data['trip_distance'].mean()
        card_share = data[data['payment_type'] == 1]['fare_amount'].sum() / total_fare if total_fare > 0 else 0
        
        # Return result
        return {
            "status": "ok",
            "metrics": {
                "trip_count": trip_count,
                "total_fare": total_fare,
                "mean_fare": mean_fare,
                "mean_trip_distance": mean_trip_distance,
                "card_share": card_share
            },
            "reason": ""
        }
    
    except Exception as e:
        return {"status": "abstain", "metrics": None, "reason": str(e)}
